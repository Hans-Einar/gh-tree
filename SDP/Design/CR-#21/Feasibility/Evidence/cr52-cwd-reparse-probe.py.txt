import ctypes,ctypes.wintypes as w,os,pathlib,struct,subprocess,sys,tempfile,json
k=ctypes.WinDLL('kernel32',use_last_error=True)
k.CreateFileW.argtypes=[w.LPCWSTR,w.DWORD,w.DWORD,ctypes.c_void_p,w.DWORD,w.DWORD,w.HANDLE];k.CreateFileW.restype=w.HANDLE
k.CloseHandle.argtypes=[w.HANDLE];k.CloseHandle.restype=w.BOOL
k.DeviceIoControl.argtypes=[w.HANDLE,w.DWORD,ctypes.c_void_p,w.DWORD,ctypes.c_void_p,w.DWORD,ctypes.POINTER(w.DWORD),ctypes.c_void_p];k.DeviceIoControl.restype=w.BOOL
INVALID=ctypes.c_void_p(-1).value
rows=[]
with tempfile.TemporaryDirectory(prefix='cr52-cwd-reparse-') as temporary:
 root=pathlib.Path(temporary);outside=root/'outside';outside.mkdir();(outside/'marker').write_text('other-directory')
 for share in [1,3]:
  for access in [0x40000000,0x100]:
   target=root/('target-'+str(share)+'-'+str(access));target.mkdir()
   guard=k.CreateFileW(str(target),0x80|0x20|0x100000,share,None,3,0x02200000,None)
   if guard==INVALID:raise ctypes.WinError(ctypes.get_last_error())
   setter=None
   try:
    setter=k.CreateFileW(str(target),access,7,None,3,0x02200000,None)
    row=dict(guardShare=share,setterAccess=hex(access),setterOpened=setter!=INVALID)
    if setter==INVALID:row['openError']=ctypes.get_last_error()
    else:
     substitute=('\\??\\'+str(outside)).encode('utf-16-le');display=str(outside).encode('utf-16-le');names=substitute+b'\0\0'+display+b'\0\0'
     data=struct.pack('<IHHHHHH',0xA0000003,8+len(names),0,0,len(substitute),len(substitute)+2,len(display))+names
     buffer=ctypes.create_string_buffer(data);used=w.DWORD();ctypes.set_last_error(0)
     ok=k.DeviceIoControl(setter,0x900A4,buffer,len(data),None,0,ctypes.byref(used),None)
     row.update(reparseSet=bool(ok),setError=0 if ok else ctypes.get_last_error())
     if ok:
      child=subprocess.run([sys.executable,'-c',"import os,json;print(json.dumps(dict(marker=open('marker').read(),cwd=os.getcwd())))"],cwd=str(target),stdout=subprocess.PIPE,check=True,universal_newlines=True)
      row['child']=json.loads(child.stdout)
    rows.append(row)
   finally:
    if setter not in (None,INVALID):k.CloseHandle(setter)
    k.CloseHandle(guard)
print(json.dumps(rows,indent=2))
