def git(p,*args,env=None,data=None,ok=True):
    r=subprocess.run(['git','-C',str(p),*args],input=data,env=env or BASE,capture_output=True)
    if ok and r.returncode:raise RuntimeError((str(p),args,r.returncode,r.stderr.decode(errors='replace')))

def text(p,*args,env=None):return git(p,*args,env=env).stdout.decode().strip()

def files(p):
    return {x.relative_to(p).as_posix():x.read_bytes() for x in p.rglob('*') if x.is_file() and '.git' not in x.relative_to(p).parts}
def snap(p):return dict(files=files(p),index=(p/'.git/index').read_bytes(),head=text(p,'rev-parse','HEAD'))
def scratch(p,label):
    s=RUN/(p.name+'-'+label);s.mkdir();fmt=text(p,'rev-parse','--show-object-format');git(s,'init','-q','--object-format='+fmt)
    env=dict(BASE,GIT_ALTERNATE_OBJECT_DIRECTORIES=str(p/'.git/objects'))
    empty=s/'.git/empty-hooks';empty.mkdir();git(s,'config','core.hooksPath',str(empty),env=env)
    for key,val in [('user.name','ScratchProbe'),('user.email','scratch@example.invalid'),('core.autocrlf','false'),('core.filemode','false'),('gc.auto','0')]:git(s,'config',key,val,env=env)
    git(s,'update-ref','--no-deref','HEAD',text(p,'rev-parse','HEAD'),env=env)
    for name,content in files(p).items():
        dst=s/name;dst.parent.mkdir(parents=True,exist_ok=True);dst.write_bytes(content)
    shutil.copyfile(p/'.git/index',s/'.git/index')
    git(s,'update-index','--really-refresh',env=env,ok=False)
    return s,env
def transferred(p,s,env,roots):
    pack=git(s,'pack-objects','--stdout','--revs',env=env,data=('\n'.join(roots)+'\n').encode()).stdout
    git(p,'index-pack','--stdin',data=pack)
    return len(pack)
def publication_index(s,env):
    entries=git(s,'ls-files','--stage','-z',env=env).stdout
    dst=s/'.git/publish-index';e=dict(env,GIT_INDEX_FILE=str(dst));git(s,'read-tree','--empty',env=e);git(s,'update-index','-z','--index-info',env=e,data=entries)
    return dst.read_bytes()
def publish(p,before,s,env,*,replace_index=True,inject=None):
    # Minimal ordinary-file feasibility model. Production path/format/journal/
    # crash/permission checks are deliberately outside this temporary probe.
    desired=files(s);changed=sorted(k for k in before['files'].keys()|desired.keys() if before['files'].get(k)!=desired.get(k))
    recovery=pathlib.Path(tempfile.mkdtemp(prefix=p.name+'-publication-',dir=RUN))
    idx=p/'.git/index';lock=p/'.git/index.lock';captured=[];installed=[]
    fd=os.open(lock,os.O_CREAT|os.O_EXCL|os.O_WRONLY|getattr(os,'O_BINARY',0))
    image=publication_index(s,env) if replace_index else b''
    os.write(fd,image);os.fsync(fd);os.close(fd)
    def finish(status,**extra):
        if lock.exists():lock.unlink()
        return dict(status=status,recovery=str(recovery),captured=captured,installed=installed,**extra)
    if idx.read_bytes()!=before['index']:return finish('index-stale')
    if inject:inject('locked',None,recovery)
    for n,name in enumerate(changed):
        path=p/name;expected=before['files'].get(name)
        if expected is None:continue
        if inject:inject('before-capture',name,recovery)
        saved=recovery/('original-'+str(n));os.rename(path,saved);captured.append((name,str(saved)))
        if saved.read_bytes()!=expected:
            for oldname,oldpath in captured:
                try:os.link(oldpath,p/oldname)
                except FileExistsError:pass
            return finish('captured-stale')
    if replace_index:
        savedidx=recovery/'original-index';os.rename(idx,savedidx)
        if savedidx.read_bytes()!=before['index']:
            try:os.link(savedidx,idx)
            except FileExistsError:pass
            for oldname,oldpath in captured:
                try:os.link(oldpath,p/oldname)
                except FileExistsError:pass
            return finish('captured-index-stale')
    if inject:inject('captured',None,recovery)
    for n,name in enumerate(changed):
        if name not in desired:continue
        payload=recovery/('installed-'+str(n));payload.write_bytes(desired[name]);(p/name).parent.mkdir(parents=True,exist_ok=True)
        if inject:inject('before-install',name,recovery)
        try:os.link(payload,p/name)
        except FileExistsError:
            if replace_index and not idx.exists():os.link(recovery/'original-index',idx)
            return finish('destination-collision')
        installed.append((name,str(payload)))
        if inject:inject('installed',name,recovery)
    if replace_index:
        os.link(lock,recovery/'installed-index')
        try:os.link(lock,idx)
        except FileExistsError:return finish('index-destination-collision')
    return finish('published')
def result(name,**details):RESULTS.append(dict(probe=name,**details))
