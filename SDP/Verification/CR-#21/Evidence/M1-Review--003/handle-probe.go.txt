package main

import (
	"os"
	"path/filepath"
	"runtime"
	"syscall"
	"testing"
	"unsafe"
)

func TestReviewerPhysicalPathHandleBalance(t *testing.T) {
	oldProcs := runtime.GOMAXPROCS(2)
	defer runtime.GOMAXPROCS(oldProcs)
	countProc := syscall.NewLazyDLL("kernel32.dll").NewProc("GetProcessHandleCount")
	process, err := syscall.GetCurrentProcess()
	if err != nil { t.Fatal(err) }
	count := func() uint32 {
		var handles uint32
		ok, _, callErr := countProc.Call(uintptr(process), uintptr(unsafe.Pointer(&handles)))
		if ok == 0 { t.Fatal(callErr) }
		return handles
	}
	dir := t.TempDir()
	file := filepath.Join(dir, "source.go")
	if err := os.WriteFile(file, []byte("package probe\n"), 0600); err != nil { t.Fatal(err) }
	if _, err := physicalPath(dir); err != nil { t.Fatal(err) }
	if _, err := physicalPath(file); err != nil { t.Fatal(err) }
	inspect := func() { for i := 0; i < 500; i++ {
		if _, err := physicalPath(dir); err != nil { t.Fatal(err) }
		if _, err := physicalPath(file); err != nil { t.Fatal(err) }
		if _, err := physicalPath(filepath.Join(dir,"missing")); err == nil { t.Fatal("missing object accepted") }
		if _, err := physicalPath("invalid\x00path"); err == nil { t.Fatal("invalid path accepted") }
	} }
	inspect()
	runtime.GC()
	before := count()
	threadsBefore, _ := runtime.ThreadCreateProfile(nil)
	for round := 0; round < 5; round++ {
		inspect()
		runtime.GC()
		after := count()
		threadsAfter,_ := runtime.ThreadCreateProfile(nil)
		t.Logf("round %d: 500 directory + 500 file + 1000 rejected path inspections: handles before=%d after=%d; threads before=%d after=%d", round,before,after,threadsBefore,threadsAfter)
		if after > before + 2 { t.Fatalf("unexpected handle growth: %d -> %d",before,after) }
	}
}
