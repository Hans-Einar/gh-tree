function Write-Probe($root, $rel, $text) {
  $dest = Join-Path $root $rel
  [IO.Directory]::CreateDirectory([IO.Path]::GetDirectoryName($dest)) | Out-Null
  [IO.File]::WriteAllText($dest, $text, $utf8)
}
