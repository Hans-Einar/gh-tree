package api_test

import (
 "strings"
 "testing"
 "time"
 a "github.com/Hans-Einar/gh-tree/internal/application/api"
 d "github.com/Hans-Einar/gh-tree/internal/domain"
)
func irvMust[T any](v T,e error) T { if e!=nil { panic(e) }; return v }
func irvSource(s string) a.SourceVersion { return irvMust(a.NewSourceVersion("git","scope","reviewer",s)) }
func irvRev(r d.RepositoryID,s string) d.Revision { return irvMust(d.NewRevision(r,irvMust(d.NewOID(strings.Repeat(s,40))))) }
func irvRepo(s string) d.RepositoryID { return irvMust(d.NewRepositoryID(d.LocalCommon,s)) }
func irvWork(r d.RepositoryID,s string) d.WorktreeID { return irvMust(d.NewWorktreeID(r,s)) }
func irvScope(w d.WorktreeID) a.WorktreeScope { return irvMust(a.NewWorktreeScope(a.WorktreeScopeData{ID:w,RootLocator:"C:/review/"+w.AdministrativeKey(),RootIdentity:irvMust(a.NewDirectoryIdentity(a.DirectoryWindows,1,[16]byte{1},"stamp")),Source:irvSource("root")})) }
func irvObservation(w d.WorktreeID) a.GitObservation { at:=time.Date(2026,9,6,0,0,0,0,time.UTC); return irvMust(a.NewGitObservation(a.GitObservationData{ID:irvMust(a.NewObservationID(w.Repository().Token()+w.AdministrativeKey())),Repository:w.Repository(),Worktree:a.Some(w),Interval:irvMust(a.NewObservationInterval(a.ObservationIntervalData{StartedAt:at,FinishedAt:at})),Version:irvSource("observation"),Completeness:a.Complete})) }
func irvFacts(w d.WorktreeID) a.WorktreeFacts { return irvMust(a.NewWorktreeFacts(a.WorktreeFactsData{ID:w,Scope:a.Some(irvScope(w)),Head:a.Some(irvMust(d.NewDetachedHead(irvRev(w.Repository(),"1")))),Availability:irvMust(a.NewAvailableWorktree(a.AvailableWorktreeData{})),Observation:irvObservation(w)})) }
func irvStatus(w d.WorktreeID) a.StatusFacts { return irvMust(a.NewStatusFacts(a.StatusFactsData{Worktree:irvFacts(w),IndexVersion:irvSource("index"),WorktreeVersion:irvSource("bytes"),ConfigurationVersion:irvSource("config"),Upstream:irvMust(a.NewNoUpstream(a.NoUpstreamData{})),Observation:irvObservation(w)})) }
func irvExpected(w d.WorktreeID) a.GitExpectedState { return irvMust(a.NewGitExpectedState(a.GitExpectedStateData{Repository:w.Repository(),Worktree:a.Some(w),Head:a.Some(irvMust(d.NewDetachedHead(irvRev(w.Repository(),"1")))),Index:a.Some(irvSource("index")),WorktreeState:a.Some(irvSource("bytes")),Observation:irvSource("status"),Configuration:irvSource("config"),Inventory:irvSource("inventory")})) }
func irvEffects() a.EffectReport { return irvMust(a.NewEffectReport(a.EffectReportData{})) }
func irvEffect(f a.EffectFacet,s a.EffectState) a.EffectReport { return irvMust(a.NewEffectReport(a.EffectReportData{Facets:[]a.FacetEffect{irvMust(a.NewFacetEffect(a.FacetEffectData{Facet:f,State:s}))}})) }
func irvEnvelope(e a.EffectReport) a.OutcomeEnvelope { return irvMust(a.NewOutcomeEnvelope(a.OutcomeEnvelopeData{Effects:e})) }
func irvTransport() a.CommandTransportOutcome { return irvMust(a.NewCommandTransportOutcome(a.CommandTransportOutcomeData{})) }
func irvDiag() a.Diagnostic { return irvMust(a.NewDiagnostic(a.DiagnosticData{Code:a.Conflict,Reason:"review",Message:"review"})) }
func irvReject(t *testing.T,e error) { t.Helper(); if e==nil {t.Fatal("contradictory value was admitted as valid")} }
func irvMutation(out a.GitMutationOutcome,kind a.GitMutationKind,e a.EffectReport) a.GitMutationResult { return irvMust(a.NewGitMutationResult(a.GitMutationResultData{Operation:irvMust(a.NewOperationID(1)),Kind:kind,PlanVersion:irvSource("plan"),Outcome:out,Transport:irvTransport(),Effects:e})) }
func irvRunVersion(w d.WorktreeID) a.StorageVersion { return irvMust(a.NewRunStorageVersion(irvScope(w),"run-store",true,20,[32]byte{1})) }
func irvSelection(w d.WorktreeID) a.DiscoveredLaunch { return irvMust(a.NewDiscoveredLaunch(a.DiscoveredLaunchData{Member:irvMust(a.NewMemberSelection(a.MemberSelectionData{LaunchPointID:irvMust(d.NewLaunchPointID(w,"npm","","dev")),SourceVersion:irvSource("manifest")}))})) }

func TestIndependentReplayGitCrossFieldScope(t *testing.T) {
 ra,rb:=irvRepo("A"),irvRepo("B"); wa,wb:=irvWork(ra,"one"),irvWork(rb,"two"); rev:=irvRev(ra,"1")
 branch:=irvMust(d.NewBranchID(ra,d.Local,"main")); ref:=irvMust(a.NewLocalBranchRef(a.LocalBranchRefData{Branch:branch}))
 t.Run("positive_ref",func(t *testing.T){if _,e:=a.NewRefFact(a.RefFactData{Locator:ref,Revision:a.Some(rev),Observation:irvObservation(wa)}); e!=nil {t.Fatal(e)}})
 t.Run("foreign_ref_revision",func(t *testing.T){_,e:=a.NewRefFact(a.RefFactData{Locator:ref,Revision:a.Some(irvRev(rb,"1")),Observation:irvObservation(wa)});irvReject(t,e)})
 t.Run("same_repo_wrong_status_worktree",func(t *testing.T){x:=irvStatus(wa).Data();x.Observation=irvObservation(irvWork(ra,"other"));_,e:=a.NewStatusFacts(x);irvReject(t,e)})
 target:=irvMust(d.NewCommitTarget(rev)); resolved:=irvMust(a.NewExactLocalResolution(a.ExactLocalResolutionData{Requested:target,Local:rev,Observation:irvObservation(wa)}))
 t.Run("created_worktree_different_target_repo",func(t *testing.T){_,e:=a.NewWorktreeCreated(a.WorktreeCreatedData{Worktree:irvFacts(wb),Target:resolved});irvReject(t,e)})
 t.Run("created_worktree_no_head_or_scope",func(t *testing.T){missing:=irvMust(a.NewWorktreeFacts(a.WorktreeFactsData{ID:wa,Availability:irvMust(a.NewMissingWorktree(a.MissingWorktreeData{})),Observation:irvObservation(wa)}));_,e:=a.NewWorktreeCreated(a.WorktreeCreatedData{Worktree:missing,Target:resolved});irvReject(t,e)})
 t.Run("stash_applied_other_repository",func(t *testing.T){stash:=irvMust(d.NewStashID(ra,rev.OID()));_,e:=a.NewStashApplied(a.StashAppliedData{Stash:stash,Status:irvStatus(wb),IndexRestored:true,Retained:true});irvReject(t,e)})
}

 wa,wb:=irvWork(irvRepo("A"),"one"),irvWork(irvRepo("B"),"two"); geom:=irvMust(a.NewGeometry(a.GeometryData{Rows:24,Columns:80})); sel:=irvSelection(wa)
 t.Run("positive_saved",func(t *testing.T){_,e:=a.NewSavedLaunch(a.SavedLaunchData{Alias:"alias",LaunchPointID:sel.Data().Member.Data().LaunchPointID,StorageVersion:irvRunVersion(wa),SourceExpectation:irvSource("manifest")});if e!=nil {t.Fatal(e)}})
 t.Run("saved_selection_foreign_run_version",func(t *testing.T){_,e:=a.NewSavedLaunch(a.SavedLaunchData{Alias:"alias",LaunchPointID:sel.Data().Member.Data().LaunchPointID,StorageVersion:irvRunVersion(wb),SourceExpectation:irvSource("manifest")});irvReject(t,e)})
 t.Run("discovery_foreign_version",func(t *testing.T){_,e:=a.NewDiscoveryRequest(a.DiscoveryRequestData{Worktree:irvScope(wa),SavedVersion:a.Some(irvRunVersion(wb))});irvReject(t,e)})
 t.Run("resolve_foreign_version",func(t *testing.T){_,e:=a.NewResolveLaunchRequest(a.ResolveLaunchRequestData{Worktree:irvScope(wa),Selection:sel,SavedVersion:a.Some(irvRunVersion(wb)),Geometry:geom});irvReject(t,e)})
 t.Run("save_foreign_version",func(t *testing.T){_,e:=a.NewSaveLaunchCommand(a.SaveLaunchCommandData{WorktreeID:wa,Selection:sel,Alias:"alias",ExpectedStorage:irvRunVersion(wb)});irvReject(t,e)})
 t.Run("default_foreign_version",func(t *testing.T){choice:=irvMust(a.NewCurrentDefaultLaunch(a.CurrentDefaultLaunchData{ExpectedStorage:a.Some(irvRunVersion(wb))}));_,e:=a.NewStartLaunchCommand(a.StartLaunchCommandData{WorktreeID:wa,Selection:choice,Geometry:geom});irvReject(t,e)})
 t.Run("retarget_exact_target_foreign_local_repository",func(t *testing.T){_,e:=a.NewRetargetWorktreeCommand(a.RetargetWorktreeCommandData{WorktreeID:wa,Target:irvMust(d.NewCommitTarget(irvRev(wb.Repository(),"1"))),Mode:irvMust(a.NewDetachRetarget(a.DetachRetargetData{})),Expected:irvExpected(wa)});irvReject(t,e)})
}

func TestIndependentReplayPublicResultSemantics(t *testing.T) {
 wa,wb:=irvWork(irvRepo("A"),"one"),irvWork(irvRepo("B"),"two"); sources:=irvMust(a.NewProjectionSources(a.ProjectionSourcesData{})); rev:=irvRev(wa.Repository(),"1"); target:=irvMust(d.NewCommitTarget(rev))
 t.Run("worktree_status_foreign_subject",func(t *testing.T){_,e:=a.NewWorktreeStatusResult(a.WorktreeStatusResultData{WorktreeID:wa,Status:a.Some(irvStatus(wb)),Sources:sources});irvReject(t,e)})
 t.Run("PR_result_accepts_commit_target",func(t *testing.T){_,e:=a.NewPullRequestDiffResult(a.PullRequestDiffResultData{Target:target,RequestedBase:irvMust(a.NewObserveCurrentPRBase(a.ObserveCurrentPRBaseData{})),Sources:sources});irvReject(t,e)})
 t.Run("deploy_resolution_replaces_exact_target",func(t *testing.T){r2:=irvRev(wa.Repository(),"2"); res:=irvMust(a.NewExactLocalResolution(a.ExactLocalResolutionData{Requested:irvMust(d.NewCommitTarget(r2)),Local:r2,Observation:irvObservation(wa)}));_,e:=a.NewDeployResult(a.DeployResultData{Target:target,Resolution:a.Some(res),Outcome:irvEnvelope(irvEffects())});irvReject(t,e)})
 refused:=irvMutation(irvMust(a.NewRefusedMutation(a.RefusedMutationData{Reason:irvDiag(),Effects:irvEffects()})),a.PushMutation,irvEffects())
 t.Run("completed_pop_wraps_refused_pushes",func(t *testing.T){_,e:=a.NewStashPopCompleted(a.StashPopCompletedData{Stash:irvMust(d.NewStashID(wa.Repository(),rev.OID())),Apply:refused,Drop:refused});irvReject(t,e)})
 t.Run("retarget_confirmation_omits_exact_target_and_worktree",func(t *testing.T){_,e:=a.NewMutationPlanSummary(a.MutationPlanSummaryData{OperationID:irvMust(a.NewOperationID(1)),Kind:a.RetargetMutation,PlanVersion:irvSource("plan"),Repository:wa.Repository(),Expected:irvExpected(wa),Choices:[]a.Choice{a.Proceed},ConfirmationRequired:true});irvReject(t,e)})
}

func TestIndependentReplayEffectNormalization(t *testing.T) {
 wa:=irvWork(irvRepo("A"),"one"); effect:=irvEffect(a.Index,a.AppliedVerified); indexed:=irvMust(a.NewIndexChanged(a.IndexChangedData{Status:irvStatus(wa),Action:a.Stage})); facts:=irvMutation(indexed,a.StageMutation,effect)
 t.Run("positive_stage",func(t *testing.T){_,e:=a.NewStageAllResult(a.StageAllResultData{Git:facts,Outcome:irvEnvelope(effect)});if e!=nil {t.Fatal(e)}})
 t.Run("command_envelope_discards_known_effect",func(t *testing.T){_,e:=a.NewStageAllResult(a.StageAllResultData{Git:facts,Outcome:irvEnvelope(irvEffects())});irvReject(t,e)})
 stage:=irvMust(a.NewStageAllResult(a.StageAllResultData{Git:facts,Outcome:irvEnvelope(effect)}))
 t.Run("terminal_says_no_change_after_applied_index",func(t *testing.T){_,e:=a.NewOperationTerminal(a.OperationTerminalData{OperationID:irvMust(a.NewOperationID(1)),Correlation:irvMust(a.NewCorrelation(a.CorrelationData{})),Disposition:a.Succeeded,Result:a.Some[a.Result](stage),Effects:irvEffect(a.Index,a.NotStarted)});irvReject(t,e)})
}
