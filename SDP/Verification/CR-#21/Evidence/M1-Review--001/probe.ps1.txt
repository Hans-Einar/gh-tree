$utf8 = [Text.UTF8Encoding]::new($false)
function Write-Probe($root, $rel, $text) {
  $dest = Join-Path $root $rel
  [IO.Directory]::CreateDirectory([IO.Path]::GetDirectoryName($dest)) | Out-Null
  [IO.File]::WriteAllText($dest, $text, $utf8)
}
function New-Probe($name, $files) {
  $root = Join-Path $reviewRoot $name
  [IO.Directory]::CreateDirectory($root) | Out-Null
  Write-Probe $root 'go.mod' "module example.test/probe`n`ngo 1.25.0`n"
  Write-Probe $root 'SDP/Design/CR-#21/MigrationMap.yaml' ([IO.File]::ReadAllText((Join-Path $candidate 'SDP/Design/CR-#21/MigrationMap.yaml')))
  Write-Probe $root 'SDP/Design/CR-#21/Verification--001.md' ([IO.File]::ReadAllText((Join-Path $candidate 'SDP/Design/CR-#21/Verification--001.md')))
  foreach ($rel in $files.Keys) { Write-Probe $root $rel $files[$rel] }
  return $root
}
