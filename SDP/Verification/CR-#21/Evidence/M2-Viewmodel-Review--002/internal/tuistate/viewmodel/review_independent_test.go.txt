func reviewMust[T any](v T, e error) T { if e != nil { panic(e) }; return v }
func reviewReject[T any](t *testing.T, v T, e error) { t.Helper(); if e == nil || !reflect.DeepEqual(v, *new(T)) { t.Fatalf("expected invalid zero %T, got error %v",v,e) } }
type reviewSpan struct { start,end uintptr; path string }
func reviewSpans(v reflect.Value, path string) (out []reviewSpan) {
 switch v.Kind() {
 case reflect.Struct:
  for i:=0;i<v.NumField();i++ { out=append(out,reviewSpans(v.Field(i),path+"."+v.Type().Field(i).Name)...)}
 case reflect.Array:
  for i:=0;i<v.Len();i++ {out=append(out,reviewSpans(v.Index(i),path+"[]")...)}
 case reflect.Slice:
  if v.Cap()>0 && v.Type().Elem().Size()>0 {out=append(out,reviewSpan{v.Pointer(),v.Pointer()+uintptr(v.Cap())*v.Type().Elem().Size(),path})}
  for i:=0;i<v.Len();i++ {out=append(out,reviewSpans(v.Index(i),path+"[]")...)}
 case reflect.Map:
  if !v.IsNil() { panic("unexpected mutable map in viewmodel") }
 }
 return out
}
func reviewDisjoint(t *testing.T,a,b any) {
 t.Helper()
 for _,x:=range reviewSpans(reflect.ValueOf(a),"a") { for _,y:=range reviewSpans(reflect.ValueOf(b),"b") { if x.start<y.end && y.start<x.end {t.Fatalf("shared mutable interval %s / %s",x.path,y.path)} } }
}
func reviewCopy(t *testing.T,spec,value any) {
 t.Helper(); t.Run(reflect.TypeOf(value).Name(),func(t *testing.T) {
  reviewDisjoint(t,spec,value)
  v:=reflect.ValueOf(value)
  if m:=v.MethodByName("Clone");m.IsValid(){ cloned:=m.Call(nil)[0].Interface();reviewDisjoint(t,value,cloned);if !reflect.DeepEqual(value,cloned){t.Fatal("clone changed value")} }
  if m:=v.MethodByName("Fields");m.IsValid(){fields:=m.Call(nil)[0].Interface();reviewDisjoint(t,value,fields);reviewDisjoint(t,fields,m.Call(nil)[0].Interface())}
 })
}
