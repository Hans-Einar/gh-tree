func irvSession(w d.WorktreeID,id uint64,phase a.SessionPhase) a.SessionSnapshot {
 cwd:=irvMust(a.NewCwdObservation(a.CwdObservationData{Worktree:irvScope(w),ProjectIdentity:irvScope(w).Data().RootIdentity,Source:irvSource("cwd")}))
 geometry:=irvMust(a.NewGeometry(a.GeometryData{Rows:24,Columns:80}))
 display:=irvMust(a.NewInvocationSummary(a.InvocationSummaryData{Cwd:cwd,AcceptedLocator:"C:/review",Terminal:a.Terminal,Geometry:geometry}))
 cleanup:=a.CleanupPending;if phase==a.Cleaned {cleanup=a.CleanupComplete}
 return irvMust(a.NewSessionSnapshot(a.SessionSnapshotData{SessionID:irvMust(d.NewSessionID(id)),WorktreeID:w,StartOperation:irvMust(a.NewOperationID(id)),Display:display,Capabilities:irvMust(a.NewSessionCapabilities(a.SessionCapabilitiesData{})),Phase:phase,Cleanup:irvMust(a.NewSessionCleanup(a.SessionCleanupData{State:cleanup})),Sequence:irvMust(a.NewSessionSequence(1)),OutputRange:irvMust(a.NewOutputRange(a.OutputRangeData{})),AcquiredCwd:a.Some(irvMust(a.NewAcquiredCwd(a.AcquiredCwdData{Observation:cwd})))}))
}
func TestFreshReplayReplayRuntimeConsistency(t *testing.T) {
 wa,wb:=irvWork(irvRepo("A"),"one"),irvWork(irvRepo("B"),"two")
 t.Run("unadmitted_start_reports_applied_resources",func(t *testing.T){_,e:=a.NewSessionStartResult(a.SessionStartResultData{Effects:irvEffect(a.RuntimeResources,a.AppliedVerified)});irvReject(t,e)})
 t.Run("running_session_reports_never_established",func(t *testing.T){_,e:=a.NewSessionStartResult(a.SessionStartResultData{Session:a.Some(irvSession(wa,1,a.Running)),Effects:irvEffects()});irvReject(t,e)})
 old:=irvMust(a.NewSessionStopResult(a.SessionStopResultData{Session:irvSession(wa,1,a.Cleaned),CleanupComplete:true,Effects:irvEffects()}))
 replaced:=irvSession(wb,2,a.Running).Data();replaced.RestartOf=a.Some(irvMust(d.NewSessionID(1)))
 replacement:=irvMust(a.NewSessionStartResult(a.SessionStartResultData{Session:a.Some(irvMust(a.NewSessionSnapshot(replaced))),Established:true,Effects:irvEffect(a.RuntimeResources,a.AppliedVerified)}))
 t.Run("restart_changes_worktree_and_cwd",func(t *testing.T){_,e:=a.NewSessionRestartResult(a.SessionRestartResultData{Old:old,Replacement:a.Some(replacement)});irvReject(t,e)})
}
