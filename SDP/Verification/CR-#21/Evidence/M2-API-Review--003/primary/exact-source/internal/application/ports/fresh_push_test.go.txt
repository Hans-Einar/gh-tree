package ports_test
import ("testing";"strings";a "github.com/Hans-Einar/gh-tree/internal/application/api";p "github.com/Hans-Einar/gh-tree/internal/application/ports";d "github.com/Hans-Einar/gh-tree/internal/domain")
func fp[T any](v T,e error)T{if e!=nil{panic(e)};return v}
func TestFreshPushEvidenceContext(t *testing.T){
 local:=fp(d.NewRepositoryID(d.LocalCommon,"independent-local"));work:=fp(d.NewWorktreeID(local,"selected"));oid:=fp(d.NewOID(strings.Repeat("a",40)));rev:=fp(d.NewRevision(local,oid));head:=fp(d.NewDetachedHead(rev));op:=fp(a.NewOperationID(99))
 sv:=func(token string)a.SourceVersion{return fp(a.NewSourceVersion("git","independent-local","fresh-review",token))};planVersion:=sv("plan")
 expected:=fp(a.NewGitExpectedState(a.GitExpectedStateData{Repository:local,Worktree:a.Some(work),Observation:sv("status"),Head:a.Some(head),Configuration:sv("config"),Inventory:sv("inventory")}))
 remoteA:=fp(d.NewRepositoryID(d.Remote,"remoteA"));remoteB:=fp(d.NewRepositoryID(d.Remote,"remoteB"));branchA:=fp(d.NewBranchID(remoteA,d.RemoteHead,"selected"));branchB:=fp(d.NewBranchID(remoteB,d.RemoteHead,"selected"));otherBranch:=fp(d.NewBranchID(remoteA,d.RemoteHead,"other"))
 binding:=func(r d.RepositoryID)a.RemoteBinding{return fp(a.NewRemoteBinding(a.RemoteBindingData{LocalRepository:local,RemoteRepository:r,RemoteName:"upstream",Configuration:sv("binding")}))};bindA,bindB:=binding(remoteA),binding(remoteB)
 id:=fp(a.NewRecoveryID("unknown-push-artifact"));subject:=fp(a.NewRecoverySubject(a.RecoverySubjectData{Repository:a.Some(remoteA),Branch:a.Some(branchA)}));record:=fp(a.NewRecoveryRecord(a.RecoveryRecordData{RecoveryID:id,Kind:a.RecoveryJournal,Layer:a.LayerGit,Subject:subject,Locator:"owned/push-journal",NextAction:"Inspect; no replay"}))
 empty:=fp(a.NewEffectReport(a.EffectReportData{}));effects:=fp(a.NewEffectReport(a.EffectReportData{Facets:[]a.FacetEffect{fp(a.NewFacetEffect(a.FacetEffectData{Facet:a.RemoteRefsPR,State:a.EffectIndeterminate,RecoveryIDs:[]a.RecoveryID{id}}))}}))
 reason:=fp(a.NewDiagnostic(a.DiagnosticData{Code:a.Indeterminate,Reason:"reply-lost",Message:"No final observation"}));partial:=fp(a.NewMutationIndeterminate(a.MutationIndeterminateData{Facts:fp(a.NewGitPostFacts(a.GitPostFactsData{})),Reason:reason,Recovery:[]a.RecoveryRecord{record},ReconciliationRequired:true}))
 facts:=fp(a.NewGitMutationResult(a.GitMutationResultData{Operation:op,Kind:a.PushMutation,PlanVersion:planVersion,Outcome:partial,Effects:effects,Recovery:[]a.RecoveryRecord{record},ReconciliationRequired:true,Transport:fp(a.NewCommandTransportOutcome(a.CommandTransportOutcomeData{}))}))
 if facts.Data().Observation.Present(){t.Fatal("fabricated post observation")}
 for _,c:=range []struct{name string;binding a.RemoteBinding;branch d.BranchID;want bool}{{"matching",bindA,branchA,true},{"foreign-repository",bindB,branchB,false},{"foreign-branch",bindA,otherBranch,false}}{t.Run(c.name,func(t *testing.T){e:=facts.ValidatePushBinding(c.binding,c.branch);if (e==nil)!=c.want{t.Fatalf("context validation=%v",e)};if facts.Data().Recovery[0].Data().RecoveryID!=id{t.Fatal("validation mutated recovery")}})}
 norm:=fp(a.NormalizeRecovery([]a.RecoveryRecord{record},nil));result:=fp(a.NewPushResult(a.PushResultData{Git:facts,RemoteEffect:effects,UpstreamEffect:empty,Outcome:fp(a.NewOutcomeEnvelope(a.OutcomeEnvelopeData{Effects:effects,Recovery:norm}))}));corr:=fp(a.NewCorrelation(a.CorrelationData{}));terminal:=fp(a.NewOperationTerminal(a.OperationTerminalData{OperationID:op,Correlation:corr,Disposition:a.Failed,Result:a.Some[a.Result](result),Effects:effects,Recovery:norm}))
 for _,c:=range []struct{b a.RemoteBinding;br d.BranchID;want bool}{{bindA,branchA,true},{bindB,branchB,false}}{request:=fp(a.NewCommandRequest(fp(a.NewPushCommand(a.PushCommandData{WorktreeID:work,Source:rev,Destination:c.br,Binding:c.b,Expected:expected})),corr));if e:=a.ValidateTerminalFor(request,terminal);(e==nil)!=c.want{t.Fatalf("original request validation=%v",e)}}
 issuer:=fp(p.NewPlanIssuer("independent-issuer"))
 for _,c:=range []struct{b a.RemoteBinding;br d.BranchID;want bool}{{bindA,branchA,true},{bindB,branchB,false}}{summary:=fp(a.NewMutationPlanSummary(a.MutationPlanSummaryData{OperationID:op,Kind:a.PushMutation,PlanVersion:planVersion,Repository:local,Worktree:a.Some(work),Head:a.Some(head),Target:a.Some(fp(d.NewCommitTarget(rev))),Expected:expected,Choices:[]a.Choice{a.Proceed},Branch:a.Some(c.br),PushBinding:a.Some(c.b)}));plan:=fp(issuer.IssuePush(p.PlanSpec{Operation:op,Version:planVersion,Token:c.br.Repository().Token(),Group:"own",Role:p.Executable,Summary:summary,SummaryDigest:[32]byte{7}}));receipt:=fp(issuer.IssueReceipt(plan,"retained",nil,false,false));wrapped,e:=p.NewExecutedGitMutation(facts,a.Some(receipt));if (e==nil)!=c.want{t.Fatalf("receipt context validation=%v",e)};if e==nil&&wrapped.Facts().Data().Recovery[0].Data().RecoveryID!=id{t.Fatal("receipt dropped recovery")}}
 bad:=facts.Data();bad.Outcome=fp(a.NewPushed(a.PushedData{Source:rev,Destination:branchB,Binding:bindB,RemoteEffect:effects,UpstreamEffect:empty}));if _,e:=a.NewGitMutationResult(bad);e==nil{t.Fatal("known returned foreign binding admitted")}
 noop:=fp(a.NewEffectReport(a.EffectReportData{Facets:[]a.FacetEffect{fp(a.NewFacetEffect(a.FacetEffectData{Facet:a.RemoteRefsPR,State:a.VerifiedNoTargetChange}))}}));good:=facts.Data();good.Outcome=fp(a.NewPushed(a.PushedData{Source:rev,Destination:branchA,Binding:bindA,RemoteEffect:noop,UpstreamEffect:empty}));good.Effects=noop;good.Recovery=nil;if _,e:=a.NewGitMutationResult(good);e!=nil{t.Fatal("no-op push refused",e)}
}
